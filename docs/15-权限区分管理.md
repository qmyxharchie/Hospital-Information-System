# 15-权限区分管理

## 基于现有功能的权限区分管理实现方案

本章节基于src目录中已经完成的功能，详细说明在医疗管理系统中实现4种身份（患者、护士、医生、管理员）权限区分管理的方案。

## 1. 概述

医疗信息系统的权限管理是保障系统安全性和数据隐私的重要组成部分。本系统采用基于角色的访问控制（RBAC）模型，通过定义不同角色及其权限来实现细粒度的访问控制。

## 2. 用户角色定义

### 2.1 患者 (PATIENT)
- **角色标识**: `PATIENT = 0`
- **主要功能**:
  - 预约挂号
  - 现场挂号
  - 查看个人挂号记录
  - 查看个人就诊历史
  - 取消个人挂号
  - 修改个人信息
- **权限限制**:
  - 仅能访问个人相关数据
  - 无法查看其他患者信息
  - 无法访问医生工作站
  - 无法修改医疗记录

### 2.2 护士 (NURSE)
- **角色标识**: `NURSE = 1`
- **主要功能**:
  - 查看科室患者列表
  - 协助医生进行叫号
  - 查看和管理候诊队列
  - 辅助挂号管理
  - 患者引导和分流
  - 基础护理记录录入
- **权限限制**:
  - 仅能查看所属科室信息
  - 不能修改医生诊断记录
  - 不能开具处方
  - 仅能取消待就诊状态的挂号

### 2.3 医生 (DOCTOR)
- **角色标识**: `DOCTOR = 2`
- **主要功能**:
  - 查看个人接诊患者列表
  - 患者叫号操作
  - 查看和更新患者病历
  - 录入诊断和治疗方案
  - 开具处方
  - 更新就诊状态
- **权限限制**:
  - 仅能管理自己接诊的患者
  - 无法访问非本人负责的患者信息
  - 需要上级医生授权才能进行特殊操作

### 2.4 管理员 (ADMIN)
- **角色标识**: `ADMIN = 3`
- **主要功能**:
  - 系统配置管理
  - 用户账户管理（创建、修改、删除）
  - 权限分配管理
  - 系统监控维护
  - 数据备份恢复
  - 日志管理
  - 系统参数设置
- **权限限制**:
  - 不能直接参与诊疗活动
  - 某些敏感操作需要双重验证
  - 操作需记录完整审计日志

```c
#ifndef DATA_STRUCTURES_H
#define DATA_STRUCTURES_H

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <time.h>

// 用户类型枚举 - 仅4种身份
typedef enum {
    PATIENT = 0,    // 患者
    NURSE = 1,      // 护士
    DOCTOR = 2,     // 医生
    ADMIN = 3       // 管理员

// 用户信息结构体
typedef struct User {
    int id;
    char name[50];
    char password[50];
    UserType type;
    char department[30];  // 科室（医生、护士、管理员专属）
    struct User* next;
    struct User* prev;
} User;

// 挂号信息结构体
typedef struct Registration {
    int reg_id;
    int patient_id;
    int doctor_id;
    char patient_name[50];
    char doctor_name[50];
    char department[30];
    time_t reg_time;        // 挂号时间
    time_t appointment_time; // 预约时间
    int status;             // 0-待就诊, 1-正在就诊, 2-已完成, 3-已取消
    struct Registration* next;
    struct Registration* prev;
} Registration;

// 候诊队列节点
typedef struct QueueNode {
    int reg_id;
    int patient_id;
    char patient_name[50];
    time_t queue_time;      // 进入队列时间
    struct QueueNode* next;
} QueueNode;

// 双向链表节点通用结构
typedef struct ListNode {
    void* data;
    struct ListNode* next;
    struct ListNode* prev;
} ListNode;

// 双向链表结构
typedef struct LinkedList {
    ListNode* head;
    ListNode* tail;
    int size;
} LinkedList;

#endif
```

### 3.3 权限验证函数实现 (permission_control.c)

```c
#include "permission_control.h"
#include <string.h>

// 检查用户是否有执行特定操作的权限
int check_permission(UserType user_type, const char* operation) {
    if (strcmp(operation, "view_own_records") == 0) {
        // 所有用户都可以查看自己的记录
        return 1;
    }
    else if (strcmp(operation, "make_appointment") == 0) {
        // 患者可以预约挂号
        return (user_type == PATIENT);
    }
    else if (strcmp(operation, "view_patient_data") == 0) {
        // 医生和护士可以查看患者数据
        return (user_type == DOCTOR || user_type == NURSE || user_type == ADMIN);
    }
    else if (strcmp(operation, "update_medical_record") == 0) {
        // 仅医生可以更新病历
        return (user_type == DOCTOR);
    }
    else if (strcmp(operation, "manage_users") == 0) {
        // 仅管理员可以管理用户
        return (user_type == ADMIN);
    }
    else if (strcmp(operation, "call_patient") == 0) {
        // 医生和护士可以叫号
        return (user_type == DOCTOR || user_type == NURSE);
    }
    else if (strcmp(operation, "cancel_registration") == 0) {
        // 患者可以取消自己的挂号，医护人员可以取消待就诊的挂号
        return (user_type == PATIENT || user_type == DOCTOR || user_type == NURSE);
    }
    else if (strcmp(operation, "access_system_config") == 0) {
        // 仅管理员可以访问系统配置
        return (user_type == ADMIN);
    }
    
    return 0; // 默认不允许
}

// 检查用户是否可以访问特定患者数据
int can_access_patient_data(UserType user_type, int requesting_user_id, int target_patient_id) {
    switch (user_type) {
        case PATIENT:
            // 患者只能访问自己的数据
            return (requesting_user_id == target_patient_id);
        case NURSE:
        case DOCTOR:
            // 医护人员可以访问他们负责的患者数据
            // 实际实现中可能需要检查科室或其他关联关系
            return 1; // 简化实现
        case ADMIN:
            // 管理员可以访问所有患者数据
            return 1;
        default:
            return 0;
    }
}

// 检查用户是否可以管理特定挂号
int can_manage_registration(UserType user_type, int requesting_user_id, int reg_patient_id) {
    switch (user_type) {
        case PATIENT:
            // 患者只能管理自己的挂号
            return (requesting_user_id == reg_patient_id);
        case NURSE:
        case DOCTOR:
            // 医护人员可以管理待就诊的挂号
            return 1; // 实际中还需要检查状态
        case ADMIN:
            // 管理员可以管理所有挂号
            return 1;
        default:
            return 0;
    }
}

// 其他权限检查函数的实现...

    return (user_type == DOCTOR);
}

int can_view_financial_data(UserType user_type) {
```

#### 3.4.2 在取消挂号中添加权限检查

在现有的 `cancelRegistration` 函数中添加权限检查：

```c
int cancelRegistration(Registration** head, Registration** tail,
    Registration* r, UserType user_type, int user_id) {
    
    if (r == NULL) {
        printf("未找到对应记录\n");
        return 0;
    }
    
    // 检查用户是否有取消挂号权限
    if (!check_permission(user_type, "cancel_registration")) {
        printf("权限不足，无法取消挂号！\n");
        return 0;
    }
    
    // 检查挂号状态，已就诊的无法取消
    if (strcmp(r->data.status, "已就诊") == 0) {
        printf("已就诊的无法取消\n");
        return 0;
    }
    
    // 如果是患者取消挂号，只能取消自己的挂号
    if (user_type == PATIENT) {
        if (!can_manage_registration(user_type, user_id, atoi(r->data.patientCardNo))) {
            printf("只能取消自己的挂号！\n");
            return 0;
        }
    }
    
    // ... (其余代码保持不变)
}
```

#### 3.4.3 在完成就诊中添加权限检查

在现有的 `completeRegistration` 函数中添加权限检查：

```c
int completeRegistration(Registration* r, UserType user_type) {
    if (r == NULL) {
        printf("挂号信息不能为空\n");
        return 0;
    }
    
    // 检查用户是否有完成就诊权限
    if (user_type != DOCTOR) {
        printf("权限不足，只有医生可以完成就诊！\n");
        return 0;
    }
    
    // 检查挂号状态是否为"待就诊"
    if (strcmp(r->data.status, "待就诊") != 0) {
        printf("挂号信息状态异常\n");
        return 0;
    }
    
    // 修改状态为"已就诊"
    strcpy(r->data.status, "已就诊");
    printf("患者%s就诊完成\n", r->data.patientName);
    return 1;
}
```

## 4. 安全措施

### 4.1 最小权限原则
- 用户仅获得完成工作所需的最小权限集
- 功能权限严格按角色划分
- 数据访问范围限制在必要范围内

### 4.2 操作审计
- 关键操作日志记录
- 用户行为轨迹追踪
- 异常操作监控与报警
- 审计日志安全存储

### 4.3 数据保护
- 敏感信息加密存储
- 数据传输加密
- 访问日志记录
- 会话超时管理

### 4.4 输入验证
- 参数合法性检查
- SQL注入防护
- 路径遍历防护
- 数据格式验证

## 5. 实现要点

### 5.1 代码实现注意事项
1. 每个涉及数据访问的函数都要进行权限验证
2. 使用统一的权限检查接口
3. 在关键操作前后记录审计日志
4. 实现权限缓存提高性能

### 5.2 与现有模块的集成
- 在登录验证模块中识别用户角色
- 在各业务模块中添加权限检查
- 在界面模块中根据角色显示相应功能

## 6. 实施建议

1. **严格遵循权限控制原则**，确保数据安全和隐私保护
2. **在关键代码处添加详细的权限检查注释**，便于维护
3. **建立权限变更审批流程**，防止权限滥用
4. **定期审查权限分配情况**，及时调整不合理权限
5. **进行权限安全测试**，确保权限控制有效

## 7. 安全最佳实践

- 定期更新密码策略
- 实施最小权限原则
- 定期进行安全审计
- 建立应急响应机制
- 进行安全意识培训
- 定期备份重要数据
- 监控异常访问行为

通过以上权限管理设计，系统能够有效控制不同角色用户的访问权限，保障医疗数据的安全性和完整性。
