# 15-权限区分管理与日志系统

## 基于现有功能的权限区分管理及日志记录实现方案

本章节基于src目录中已经完成的功能，详细说明在医疗管理系统中实现4种身份（患者、护士、医生、管理员）权限区分管理的方案，并添加完整的日志记录功能，完全基于现有的结构体定义。

## 1. 概述

医疗信息系统的权限管理是保障系统安全性和数据隐私的重要组成部分。本系统采用基于角色的访问控制（RBAC）模型，通过定义不同角色及其权限来实现细粒度的访问控制，并通过日志系统记录所有关键操作。

## 3. 日志系统设计

### 3.1 日志系统头文件 (log.h)
```c
//===============
//    日志系统
//===============

/*日志系统提供操作记录功能
- 记录关键操作
- 记录用户行为
- 记录系统事件
- 日志文件管理*/

#ifndef _LOG_H_
#define _LOG_H_

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <time.h>

// 日志级别枚举
typedef enum {
    LOG_DEBUG = 0,    // 调试信息
    LOG_INFO = 1,     // 一般信息
    LOG_WARN = 2,     // 警告信息
    LOG_ERROR = 3     // 错误信息
} LogLevel;

// 日志记录结构体
typedef struct LogEntry {
    char timestamp[20];    // 时间戳
    LogLevel level;        // 日志级别
    char module[50];       // 模块名称
    char message[500];     // 日志消息
    struct LogEntry* next;
} LogEntry;

// 日志函数声明
void initLogging(void);                                    // 初始化日志系统
void logMessage(LogLevel level, const char* module, const char* format, ...);  // 记录日志消息
void logOperation(const char* username, const char* operation, const char* details);  // 记录操作日志
void logUserLogin(const char* username);                  // 记录用户登录
void logUserLogout(const char* username);                 // 记录用户登出
void logSecurityEvent(const char* event, const char* details);  // 记录安全事件
void cleanupLogging(void);                               // 清理日志系统

// 便捷宏定义
#define LOG_DEBUG_MSG(module, ...) logMessage(LOG_DEBUG, module, __VA_ARGS__)
#define LOG_INFO_MSG(module, ...) logMessage(LOG_INFO, module, __VA_ARGS__)
#define LOG_WARN_MSG(module, ...) logMessage(LOG_WARN, module, __VA_ARGS__)
#define LOG_ERROR_MSG(module, ...) logMessage(LOG_ERROR, module, __VA_ARGS__)

#endif
```

### 3.2 日志系统实现 (log.c)
```c
#include "log.h"
#include <stdarg.h>
#include <sys/stat.h>

// 日志文件路径
#define LOG_FILE_PATH "data/log.txt"

// 全局日志文件指针
static FILE* g_logFile = NULL;

// 初始化日志系统
void initLogging(void) {
    // 确保数据目录存在
    #ifdef _WIN32
        mkdir("data");
    #else
        mkdir("data", 0755);
    #endif
    
    // 以追加模式打开日志文件
    g_logFile = fopen(LOG_FILE_PATH, "a");
    if (g_logFile == NULL) {
        printf("警告：无法打开日志文件 %s\n", LOG_FILE_PATH);
        return;
    }
    
    // 记录系统启动日志
    time_t now = time(0);
    char* time_str = ctime(&now);
    time_str[strlen(time_str)-1] = '\0'; // 移除换行符
    
    fprintf(g_logFile, "[%s] [INFO] [SYSTEM] 日志系统初始化\n", time_str);
    fflush(g_logFile);
}

// 获取当前时间字符串
void getCurrentTimeString(char* buffer, size_t bufferSize) {
    time_t now = time(0);
    struct tm* timeinfo = localtime(&now);
    strftime(buffer, bufferSize, "%Y-%m-%d %H:%M:%S", timeinfo);
}

// 记录日志消息
void logMessage(LogLevel level, const char* module, const char* format, ...) {
    if (g_logFile == NULL) {
        // 如果日志文件未初始化，尝试重新初始化
        initLogging();
        if (g_logFile == NULL) {
            return; // 如果还是失败，则放弃记录
        }
    }
    
    char timestamp[20];
    getCurrentTimeString(timestamp, sizeof(timestamp));
    
    // 获取日志级别字符串
    const char* levelStr;
    switch(level) {
        case LOG_DEBUG: levelStr = "DEBUG"; break;
        case LOG_INFO:  levelStr = "INFO";  break;
        case LOG_WARN:  levelStr = "WARN";  break;
        case LOG_ERROR: levelStr = "ERROR"; break;
        default:        levelStr = "UNKNOWN"; break;
    }
    
    // 格式化日志消息
    char message[500];
    va_list args;
    va_start(args, format);
    vsnprintf(message, sizeof(message), format, args);
    va_end(args);
    
    // 写入日志文件
    fprintf(g_logFile, "[%s] [%s] [%s] %s\n", timestamp, levelStr, module, message);
    fflush(g_logFile);
}

// 记录操作日志
void logOperation(const char* username, const char* operation, const char* details) {
    if (g_logFile == NULL) {
        initLogging();
        if (g_logFile == NULL) {
            return;
        }
    }
    
    char timestamp[20];
    getCurrentTimeString(timestamp, sizeof(timestamp));
    
    fprintf(g_logFile, "[%s] [INFO] [OPERATION] 用户[%s] 执行操作[%s] - %s\n", 
            timestamp, username, operation, details ? details : "");
    fflush(g_logFile);
}

// 记录用户登录
void logUserLogin(const char* username) {
    if (g_logFile == NULL) {
        initLogging();
        if (g_logFile == NULL) {
            return;
        }
    }
    
    char timestamp[20];
    getCurrentTimeString(timestamp, sizeof(timestamp));
    
    fprintf(g_logFile, "[%s] [INFO] [AUTH] 用户[%s] 登录系统\n", timestamp, username);
    fflush(g_logFile);
}

// 记录用户登出
void logUserLogout(const char* username) {
    if (g_logFile == NULL) {
        initLogging();
        if (g_logFile == NULL) {
            return;
        }
    }
    
    char timestamp[20];
    getCurrentTimeString(timestamp, sizeof(timestamp));
    
    fprintf(g_logFile, "[%s] [INFO] [AUTH] 用户[%s] 登出系统\n", timestamp, username);
    fflush(g_logFile);
}

// 记录安全事件
void logSecurityEvent(const char* event, const char* details) {
    if (g_logFile == NULL) {
        initLogging();
        if (g_logFile == NULL) {
            return;
        }
    }
    
    char timestamp[20];
    getCurrentTimeString(timestamp, sizeof(timestamp));
    
    fprintf(g_logFile, "[%s] [WARN] [SECURITY] 安全事件[%s] - %s\n", 
            timestamp, event, details ? details : "");
    fflush(g_logFile);
}

// 清理日志系统
void cleanupLogging(void) {
    if (g_logFile != NULL) {
        time_t now = time(0);
        char* time_str = ctime(&now);
        time_str[strlen(time_str)-1] = '\0'; // 移除换行符
        
        fprintf(g_logFile, "[%s] [INFO] [SYSTEM] 日志系统关闭\n\n", time_str);
        fflush(g_logFile);
        fclose(g_logFile);
        g_logFile = NULL;
    }
}
```

## 4. 角色映射

由于现有系统中role字段只区分普通用户和管理员(0=用户, 1=管理员)，我们需要在此基础上扩展角色概念：

```c
// 在新的权限管理中，我们基于用户名和预设的角色分配来确定具体角色
typedef enum {
    PATIENT_ROLE = 0,    // 患者（对应原系统中的普通用户）
    NURSE_ROLE = 1,      // 护士（扩展角色）
    DOCTOR_ROLE = 2,     // 医生（扩展角色）
    ADMIN_ROLE = 3       // 管理员（对应原系统中的管理员）
} ExtendedRole;
```

## 8. 安全措施

### 8.1 最小权限原则
- 用户仅获得完成工作所需的最小权限集
- 功能权限严格按角色划分
- 数据访问范围限制在必要范围内

### 8.2 操作审计
- 关键操作日志记录
- 用户行为轨迹追踪
- 异常操作监控与报警
- 审计日志安全存储

### 8.3 数据保护
- 敏感信息加密存储
- 访问日志记录
- 会话超时管理

### 8.4 输入验证
- 参数合法性检查
- 数据格式验证

## 9. 日志文件管理

- 系统启动和关闭事件
