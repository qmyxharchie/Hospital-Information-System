# 15-权限区分管理

## 4种身份（患者、护士、医生、管理员）的权限区分管理实现方案

本章节详细说明在医疗管理系统中实现4种身份（患者、护士、医生、管理员）权限区分管理所需的具体代码实现。

## 1. 概述

医疗信息系统的权限管理是保障系统安全性和数据隐私的重要组成部分。本系统采用基于角色的访问控制（RBAC）模型，通过定义不同角色及其权限来实现细粒度的访问控制。

## 2. 用户角色定义

### 2.1 患者 (PATIENT)
- **角色标识**: `PATIENT = 0`
- **主要功能**:
  - 预约挂号
  - 现场挂号
  - 查看个人挂号记录
  - 查看个人就诊历史
  - 取消个人挂号
  - 修改个人信息
- **权限限制**:
  - 仅能访问个人相关数据
  - 无法查看其他患者信息
  - 无法访问医生工作站
  - 无法修改医疗记录

### 2.2 护士 (NURSE)
- **角色标识**: `NURSE = 1`
- **主要功能**:
  - 查看科室患者列表
  - 协助医生进行叫号
  - 查看和管理候诊队列
  - 辅助挂号管理
  - 患者引导和分流
  - 基础护理记录录入
- **权限限制**:
  - 仅能查看所属科室信息
  - 不能修改医生诊断记录
  - 不能开具处方
  - 仅能取消待就诊状态的挂号

### 2.3 医生 (DOCTOR)
- **角色标识**: `DOCTOR = 2`
- **主要功能**:
  - 查看个人接诊患者列表
  - 患者叫号操作
  - 查看和更新患者病历
  - 录入诊断和治疗方案
  - 开具处方
  - 更新就诊状态
- **权限限制**:
  - 仅能管理自己接诊的患者
  - 无法访问非本人负责的患者信息
  - 需要上级医生授权才能进行特殊操作

### 2.4 管理员 (ADMIN)
- **角色标识**: `ADMIN = 3`
- **主要功能**:
  - 系统配置管理
  - 用户账户管理（创建、修改、删除）
  - 权限分配管理
  - 系统监控维护
  - 数据备份恢复
  - 日志管理
  - 系统参数设置
- **权限限制**:
  - 不能直接参与诊疗活动
  - 某些敏感操作需要双重验证
  - 操作需记录完整审计日志

```c
    int reg_id;

    char department[30];

typedef struct ListNode {
    void* data;

    struct ListNode* prev;

typedef struct LinkedList {
    ListNode* head;
```

### 3.3 权限验证函数实现 (permission_control.c)

```c
#include "permission_control.h"
#include <string.h>

// 检查用户是否有执行特定操作的权限
int check_permission(UserType user_type, const char* operation) {
    if (strcmp(operation, "view_own_records") == 0) {
        // 所有用户都可以查看自己的记录
        return 1;
    }
    else if (strcmp(operation, "make_appointment") == 0) {
        // 患者可以预约挂号
        return (user_type == PATIENT);
    }
    else if (strcmp(operation, "view_patient_data") == 0) {
        // 医生和护士可以查看患者数据
        return (user_type == DOCTOR || user_type == NURSE || user_type == ADMIN);
    }
    else if (strcmp(operation, "update_medical_record") == 0) {
        // 仅医生可以更新病历
        return (user_type == DOCTOR);
    }
    else if (strcmp(operation, "manage_users") == 0) {
        // 仅管理员可以管理用户
        return (user_type == ADMIN);
    }
    else if (strcmp(operation, "call_patient") == 0) {
        // 医生和护士可以叫号
        return (user_type == DOCTOR || user_type == NURSE);
    }
    else if (strcmp(operation, "cancel_registration") == 0) {
        // 患者可以取消自己的挂号，医护人员可以取消待就诊的挂号
        return (user_type == PATIENT || user_type == DOCTOR || user_type == NURSE);
    }
    else if (strcmp(operation, "access_system_config") == 0) {
        // 仅管理员可以访问系统配置
        return (user_type == ADMIN);
    }
    
    return 0; // 默认不允许
}

// 检查用户是否可以访问特定患者数据
int can_access_patient_data(UserType user_type, int requesting_user_id, int target_patient_id) {
    switch (user_type) {
        case PATIENT:
            // 患者只能访问自己的数据
            return (requesting_user_id == target_patient_id);
        case NURSE:
        case DOCTOR:
            // 医护人员可以访问他们负责的患者数据
            // 实际实现中可能需要检查科室或其他关联关系
            return 1; // 简化实现
        case ADMIN:
            // 管理员可以访问所有患者数据
            return 1;
        default:
            return 0;
    }
}

// 检查用户是否可以管理特定挂号
int can_manage_registration(UserType user_type, int requesting_user_id, int reg_patient_id) {
    switch (user_type) {
        case PATIENT:
            // 患者只能管理自己的挂号
            return (requesting_user_id == reg_patient_id);
        case NURSE:
        case DOCTOR:
            // 医护人员可以管理待就诊的挂号
            return 1; // 实际中还需要检查状态
        case ADMIN:
            // 管理员可以管理所有挂号
            return 1;
        default:
            return 0;
    }
}
```

```c
int cancel_registration(int reg_id, int user_id, UserType user_type) {
    Registration* current = g_registrations;
    while (current) {
        if (current->reg_id == reg_id) {
            // 检查权限：患者只能取消自己的挂号，医护人员可以取消待就诊的挂号
            if ((user_type == PATIENT && current->patient_id == user_id) ||
                ((user_type == DOCTOR || user_type == NURSE) && current->status == 0)) {
                
                // 执行取消操作
                current->status = 3; // 已取消
                
                // 保存更改
                save_registrations_to_file(g_registrations);
                
                return 0; // 成功
            } else {
                return -1;  // 权限不足
            }
        }
        current = current->next;
    }
    return -1;  // 未找到挂号记录
}
```
