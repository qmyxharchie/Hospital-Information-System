### 3.2 CRUD操作实现

#### 添加操作示例（以Medicine为例）
```c
void addMedicine(Medicine** head, Medicine** tail,
                char* genericName, char* brandName, char* spec,
                double price, int stock, int minStock) {
    Medicine* newNode = (Medicine*)malloc(sizeof(Medicine));
    
    // 初始化节点
    newNode->next = newNode->pre = NULL;
    
    // 生成唯一ID
    char id[20];
    generateUniqueId("MED", id);
    
    // 填充数据
    strcpy(newNode->data.medNo, id);
    strcpy(newNode->data.genericName, genericName);
    strcpy(newNode->data.brandName, brandName);
    strcpy(newNode->data.spec, spec);
    newNode->data.price = price;
    newNode->data.stock = stock;
    newNode->data.minStock = minStock;
    
    // 添加到链表尾部
    if (*tail == NULL) {
        *head = *tail = newNode;
    } else {
        (*tail)->next = newNode;
        newNode->pre = *tail;
        *tail = newNode;
    }
    
    // 保存到文件
    rebuildMedicineFile(*head);
}
```

## 5. 安全性和稳定性增强

### 5.1 输入验证增强

```c
// 安全字符串复制
void safeStringCopy(char* dest, const char* src, int maxLen) {
    if (dest == NULL || maxLen <= 0) return;
    if (src == NULL) {
        dest[0] = '\0';
        return;
    }
    strncpy(dest, src, maxLen - 1);
    dest[maxLen - 1] = '\0';
}

// 验证手机号
bool isValidPhone(const char* phone) {
    if (phone == NULL || strlen(phone) != 11) {
        return false;
    }
    for (int i = 0; i < 11; i++) {
        if (!isdigit(phone[i])) {
            return false;
        }
    }
    return true;
}

// 验证身份证号
bool isValidIdCard(const char* idCard) {
    if (idCard == NULL || strlen(idCard) != 18) {
        return false;
    }
    for (int i = 0; i < 17; i++) {
        if (!isdigit(idCard[i])) {
            return false;
        }
    }
    char last = idCard[17];
    return isdigit(last) || last == 'X' || last == 'x';
}
```

void freePatientChain(Patient* head) {
    Patient* current = head;
    while (current != NULL) {
        Patient* next = current->next;
        free(current);
        current = next;
    }
}
