# 挂号系统详细实现 - 预约挂号与现场挂号（更新版）

## 1. 挂号系统概述

挂号系统是医疗信息系统的核心模块之一，负责管理患者与医生之间的预约关系。系统支持两种挂号方式：
- 预约挂号：提前预约未来的就诊时间
- 现场挂号：当天即时挂号

## 4. 挂号功能实现

### 4.1 预约挂号实现

```c
/**
 * 预约挂号函数
 * @param head 挂号链表头指针的指针
 * @param tail 挂号链表尾指针的指针
 * @param patientCardNo 患者卡号
 * @param patientName 患者姓名
 * @param doctorEmpNo 医生工号
 * @param doctorName 医生姓名
 * @param dept 科室
 * @param appointmentDate 预约日期 (YYYY-MM-DD)
 * @param appointmentTime 预约时间 (HH:MM)
 * @param consultationFee 诊疗费用
 * @param remarks 备注信息
 * @param createdBy 挂号创建者角色类型
 * @return 成功返回1，失败返回0
 */
int addRegistration(Registration** head, Registration** tail,
    char* patientCardNo, char* patientName,
    char* doctorEmpNo, char* doctorName, char* dept,
    char* appointmentDate, char* appointmentTime, 
    float consultationFee, char* remarks, UserRole createdBy) {
    
    // 1. 验证医生是否存在
    Doctor* d = findDoctorByEmpNo(getDoctorHead(), doctorEmpNo);
    if (d == NULL) {
        printf("未找到对应医生\n");
        return 0;
    }
    
    // 2. 检查医生是否还有名额
    if (!canAcceptPatient(d)) {
        printf("该医生当日号源已满\n");
        return 0;
    }
    
    // 3. 检查同一名患者是否已有同一天同一医生的待就诊挂号
    Registration* temp = *head;
    while(temp != NULL) {
        if(strcmp(temp->data.patientCardNo, patientCardNo) == 0 && 
           strcmp(temp->data.doctorEmpNo, doctorEmpNo) == 0 &&
           strcmp(temp->data.appointmentDate, appointmentDate) == 0 &&
           (temp->data.status == REG_PENDING || temp->data.status == REG_IN_PROGRESS)) {
            printf("该患者在该医生处已有待就诊或就诊中的挂号记录\n");
            return 0;
        }
        temp = temp->next;
    }
    
    // 4. 分配新节点
    Registration* newNode = (Registration*)malloc(sizeof(Registration));
    if (newNode == NULL) {
        printf("内存分配失败\n");
        return 0;
    }
    
    // 5. 初始化节点指针
    newNode->next = newNode->pre = NULL;
    
    // 6. 生成挂号单号
    char id[20];
    generateUniqueId("REG", id);
    strcpy(newNode->data.regNo, id);
    
    // 7. 填充挂号信息
    strcpy(newNode->data.patientCardNo, patientCardNo);
    strcpy(newNode->data.patientName, patientName);
    strcpy(newNode->data.doctorEmpNo, doctorEmpNo);
    strcpy(newNode->data.doctorName, doctorName);
    strcpy(newNode->data.dept, dept);
    
    // 8. 获取当前日期和时间作为挂号时间
    int year, month, day, hour, minute, second;
    time_t now = time(0);
    struct tm* timeinfo = localtime(&now);
    
    year = timeinfo->tm_year + 1900;
    month = timeinfo->tm_mon + 1;
    day = timeinfo->tm_mday;
    hour = timeinfo->tm_hour;
    minute = timeinfo->tm_min;
    second = timeinfo->tm_sec;
    
    sprintf(newNode->data.date, "%04d-%02d-%02d", year, month, day);
    sprintf(newNode->data.time, "%02d:%02d:%02d", hour, minute, second);
    
    // 9. 设置预约日期和时间
    strcpy(newNode->data.appointmentDate, appointmentDate);
    strcpy(newNode->data.appointmentTime, appointmentTime);
    
    // 10. 设置初始状态为待就诊
    newNode->data.status = REG_PENDING;
    
    // 11. 记录挂号创建者角色
    newNode->data.createdBy = createdBy;
    
    // 12. 设置费用和备注信息（保留字段）
    newNode->data.consultationFee = consultationFee;
    strcpy(newNode->data.remarks, remarks);
    
    // 13. 尾插法插入链表
    if (*tail == NULL) {
        *head = newNode;
        *tail = newNode;
    } else {
        (*tail)->next = newNode;
        newNode->pre = *tail;
        *tail = newNode;
    }
    
    // 14. 增加医生的接诊人数（仅对当天的预约）
    if(strcmp(appointmentDate, newNode->data.date) == 0) {
        // 如果是当天挂号，则增加当天接诊人数
        incrementPatientCount(d);
    }
    
    // 15. 保存到文件
    rebuildRegistrationFile(*head);
    
    // 16. 输出挂号成功信息
    if(createdBy == PATIENT) {
        printf("预约挂号成功！挂号编号：%s，患者：%s，医生：%s，科室：%s，预约日期：%s %s，费用：%.2f元，备注：%s\n",
            newNode->data.regNo, newNode->data.patientName,
            newNode->data.doctorName, newNode->data.dept, 
            newNode->data.appointmentDate, newNode->data.appointmentTime, 
            newNode->data.consultationFee, newNode->data.remarks);
    } else {
        printf("现场挂号成功！挂号编号：%s，患者：%s，医生：%s，科室：%s，挂号日期：%s，费用：%.2f元，备注：%s\n",
            newNode->data.regNo, newNode->data.patientName,
            newNode->data.doctorName, newNode->data.dept, newNode->data.date, 
            newNode->data.consultationFee, newNode->data.remarks);
    }
    
    return 1;
}
```

### 4.2 预约挂号界面实现

```c
/**
 * 预约挂号界面
 * 由患者操作，用于预约未来的就诊时间
 */
void patientAppointmentRegistration() {
    char patientCardNo[20], patientName[50];
    char doctorEmpNo[20], doctorName[50], dept[50];
    char appointmentDate[20], appointmentTime[20];
    float consultationFee;
    char remarks[100];
    int success;
    
    printf("=== 患者预约挂号 ===\n");
    
    // 1. 获取患者信息
    printf("请输入您的卡号: ");
    scanf("%s", patientCardNo);
    
    // 从患者链表中获取患者信息
    Patient* patient = findPatientByCardNo(getPatientHead(), patientCardNo);
    if(patient == NULL) {
        printf("未找到对应患者信息\n");
        return;
    }
    strcpy(patientName, patient->data.name);
    
    // 2. 获取医生信息
    printf("请输入医生工号: ");
    scanf("%s", doctorEmpNo);
    
    // 从医生链表中获取医生信息
    Doctor* doctor = findDoctorByEmpNo(getDoctorHead(), doctorEmpNo);
    if(doctor == NULL) {
        printf("未找到对应医生\n");
        return;
    }
    strcpy(doctorName, doctor->data.name);
    strcpy(dept, doctor->data.dept);
    
    // 3. 获取预约日期
    printf("请输入预约日期 (YYYY-MM-DD，如今天请输入 %04d-%02d-%02d): ", 
           g_currentYear, g_currentMonth, g_currentDay);
    scanf("%s", appointmentDate);
    
    // 4. 验证日期有效性
    int year, month, day;
    if(sscanf(appointmentDate, "%d-%d-%d", &year, &month, &day) != 3) {
        printf("日期格式错误\n");
        return;
    }
    
    if(!isValidDate(year, month, day)) {
        printf("无效的日期\n");
        return;
    }
    
    // 5. 检查预约日期不能早于今天
    if(compareDate(year, month, day, g_currentYear, g_currentMonth, g_currentDay) < 0) {
        printf("预约日期不能早于今天\n");
        return;
    }
    
    // 6. 获取预约时间
    printf("请输入预约时间 (HH:MM，如 09:00): ");
    scanf("%s", appointmentTime);
    
    // 7. 验证时间格式
    int hour, minute;
    if(sscanf(appointmentTime, "%d:%d", &hour, &minute) != 2) {
        printf("时间格式错误\n");
        return;
    }
    
    if(hour < 0 || hour > 23 || minute < 0 || minute > 59) {
        printf("时间无效\n");
        return;
    }
    
    // 8. 获取费用和备注信息
    printf("请输入诊疗费用: ");
    scanf("%f", &consultationFee);
    printf("请输入备注信息: ");
    scanf("%s", remarks);
    
    // 9. 执行预约挂号
    success = addRegistration(&g_registrationHead, &g_registrationTail,
                             patientCardNo, patientName,
                             doctorEmpNo, doctorName, dept,
                             appointmentDate, appointmentTime,
                             consultationFee, remarks, PATIENT); // 使用正确的UserRole参数
    
    if(success) {
        printf("预约挂号成功！请按时就诊。\n");
    } else {
        printf("预约挂号失败！\n");
    }
}
```

```c
        printf("未找到对应医生\n");
        return;
    }
    strcpy(doctorName, doctor->data.name);
    strcpy(dept, doctor->data.dept);
    
    // 3. 设置为当天日期和时间
    int year, month, day, hour, minute, second;
    time_t now = time(0);
    struct tm* timeinfo = localtime(&now);
    
    year = timeinfo->tm_year + 1900;
    month = timeinfo->tm_mon + 1;
    day = timeinfo->tm_mday;
    hour = timeinfo->tm_hour;
    minute = timeinfo->tm_min;
    second = timeinfo->tm_sec;

    scanf("%f", &consultationFee);
    printf("请输入备注信息: ");
    scanf("%s", remarks);
    
    // 5. 执行现场挂号

                            patientCardNo, patientName,
```

```c
int patientCancelRegistration(char* patientCardNo, char* regNo) {
    Registration* head = getRegistrationHead();
    Registration* current = head;
    
    // 查找指定的挂号记录
    while(current != NULL) {
        if(strcmp(current->data.patientCardNo, patientCardNo) == 0 &&
           strcmp(current->data.regNo, regNo) == 0) {
            break;
        }
        current = current->next;
    }
    
    if(current == NULL) {
        printf("未找到对应的挂号记录\n");
        return 0;
    }
    
    // 检查状态，只有待就诊的挂号才能取消
    if(current->data.status != REG_PENDING) {
        printf("只有待就诊状态的挂号才能取消\n");
        return 0;
    }
    
    // 查找对应医生并减少其当前患者数量（如果是当天的挂号）
    if(strcmp(current->data.appointmentDate, current->data.date) == 0) {
        Doctor* d = findDoctorByEmpNo(getDoctorHead(), current->data.doctorEmpNo);
        if(d != NULL && d->data.currentPatients > 0) {
            d->data.currentPatients--;
        }
    }
    
    // 修改挂号记录状态为已取消
    current->data.status = REG_CANCELLED;
    
    // 更新挂号记录文件
    rebuildRegistrationFile(head);
    printf("挂号取消成功\n");
    return 1;
}
```

```c
/**
 * 按状态查询挂号记录
 */
Registration* findRegistrationsByStatus(Registration* head, RegStatus status) {
    Registration* current = head;
    Registration* resultHead = NULL;
    Registration* resultTail = NULL;
    
    while(current != NULL) {
        if(current->data.status == status) {
            // 创建新节点并复制数据
            Registration* newNode = (Registration*)malloc(sizeof(Registration));
            if(newNode != NULL) {
                newNode->data = current->data;
                newNode->next = NULL;
                
                if(resultHead == NULL) {
                    resultHead = newNode;
                    resultTail = newNode;
                } else {
                    resultTail->next = newNode;
                    resultTail = newNode;
                }
            }
        }
        current = current->next;
    }
    
    return resultHead;
}
```
