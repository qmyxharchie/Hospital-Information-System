# 14-挂号系统 - 详细实现指南

## 1. 模块概述

### 1.1 功能描述
挂号模块负责管理病人的挂号记录：
- 预约挂号（自动检查医生名额）
- 现场挂号
- 取消挂号
- 完成就诊
- 时间范围查询
- 挂号队列管理

### 1.2 业务规则
1. 病人必须存在（应先添加病人）
2. 医生必须有剩余名额（currentPatients < maxPatients）
3. 同一医生同时只能有一个"待就诊"状态的挂号
4. 挂号成功后自动增加医生的currentPatients计数
5. 支持按不同条件查询挂号记录

## 2. 数据结构

在 `registration.h` 中定义 Registration 结构体：

### 2.1 RegistrationData 结构体
```c
typedef struct RegistrationData {
    char regNo[20];              // 挂号单号（如"REG001"）
    char patientCardNo[20];      // 病人卡号
    char patientName[50];        // 病人姓名（冗余存储，便于显示）
    char doctorEmpNo[20];        // 医生工号
    char doctorName[50];         // 医生姓名（冗余存储）
    char dept[50];               // 科室
    char date[20];               // 挂号日期（格式：YYYY-MM-DD）
    char time[20];               // 挂号时间（格式：HH:MM:SS）
    char status[20];             // 状态（待就诊/就诊中/已完成/已取消）
} RegistrationData;
```

### 2.2 Registration 链表节点结构体
```c
typedef struct Registration {
    RegistrationData data;
    struct Registration* next;
    struct Registration* pre;
} Registration;
```

## 3. 核心函数实现

### 3.1 从文件加载挂号记录
```c
void buildRegistrationChain(Registration** head, Registration** tail) {
    FILE* fp = fopen(REGISTRATION_FILE, "r");
    if (fp == NULL) {
        // 文件不存在，创建空文件
        fp = fopen(REGISTRATION_FILE, "w");
        if (fp != NULL) {
            fprintf(fp, "-1\n");
            fclose(fp);
        }
        *head = NULL;
        *tail = NULL;
        return;
    }
    
    int opt;
    Registration* pre = NULL;
    
    while (1) {
        if (fscanf(fp, "%d", &opt) != 1) break;
        if (opt == -1) break;
        
        Registration* node = (Registration*)malloc(sizeof(Registration));
        if (node == NULL) {
            printf("内存分配失败\n");
            break;
        }
        
        node->next = node->pre = NULL;
        
        // 依次读取各字段
        fscanf(fp, "%s", node->data.regNo);
        fscanf(fp, "%s", node->data.patientCardNo);
        fscanf(fp, "%s", node->data.patientName);
        fscanf(fp, "%s", node->data.doctorEmpNo);
        fscanf(fp, "%s", node->data.doctorName);
        fscanf(fp, "%s", node->data.dept);
        fscanf(fp, "%s", node->data.date);
        fscanf(fp, "%s", node->data.time);
        fscanf(fp, "%s", node->data.status);
        
        // 跳过空行
        fgetc(fp);
        
        // 尾插法插入链表
        if (*head == NULL) {
            *head = node;
        } else {
            pre->next = node;
            node->pre = pre;
        }
        pre = node;
    }
    
    *tail = pre;
    fclose(fp);
}
```

### 3.2 挂号（最核心的函数）
```c
int addRegistration(Registration** head, Registration** tail,
    char* patientCardNo, char* patientName,
    char* doctorEmpNo, char* doctorName, char* dept) {
    
    // 1. 验证医生是否存在
    Doctor* d = findDoctorByEmpNo(getDoctorHead(), doctorEmpNo);
    if (d == NULL) {
        printf("未找到对应医生\n");
        return 0;
    }
    
    // 2. 检查医生是否还有名额
    if (!canAcceptPatient(d)) {
        printf("该医生今日号已满\n");
        return 0;
    }
    
    // 3. 检查同一名患者是否已有待就诊的挂号
    Registration* temp = *head;
    while(temp != NULL) {
        if(strcmp(temp->data.patientCardNo, patientCardNo) == 0 && 
           strcmp(temp->data.status, "待就诊") == 0 &&
           strcmp(temp->data.doctorEmpNo, doctorEmpNo) == 0) {
            printf("该患者在该医生处已有待就诊的挂号\n");
            return 0;
        }
        temp = temp->next;
    }
    
    // 4. 分配新节点
    Registration* newNode = (Registration*)malloc(sizeof(Registration));
    if (newNode == NULL) {
        printf("内存分配失败\n");
        return 0;
    }
    
    // 5. 初始化节点指针
    newNode->next = newNode->pre = NULL;
    
    // 6. 生成挂号单号
    char id[20];
    generateUniqueId("REG", id);
    strcpy(newNode->data.regNo, id);
    
    // 7. 填充挂号信息
    strcpy(newNode->data.patientCardNo, patientCardNo);
    strcpy(newNode->data.patientName, patientName);
    strcpy(newNode->data.doctorEmpNo, doctorEmpNo);
    strcpy(newNode->data.doctorName, doctorName);
    strcpy(newNode->data.dept, dept);
    
    // 8. 获取当前日期和时间
    int year, month, day, hour, minute, second;
    time_t now = time(0);
    struct tm* timeinfo = localtime(&now);
    
    year = timeinfo->tm_year + 1900;
    month = timeinfo->tm_mon + 1;
    day = timeinfo->tm_mday;
    hour = timeinfo->tm_hour;
    minute = timeinfo->tm_min;
    second = timeinfo->tm_sec;
    
    sprintf(newNode->data.date, "%04d-%02d-%02d", year, month, day);
    sprintf(newNode->data.time, "%02d:%02d:%02d", hour, minute, second);
    
    // 9. 设置初始状态
    strcpy(newNode->data.status, "待就诊");
    
    // 10. 尾插法插入链表
    if (*tail == NULL) {
        *head = newNode;
        *tail = newNode;
    } else {
        (*tail)->next = newNode;
        newNode->pre = *tail;
        *tail = newNode;
    }
    
    // 11. 增加医生的接诊人数
    incrementPatientCount(d);
    
    // 12. 保存到文件
    rebuildRegistrationFile(*head);
    
    // 13. 输出挂号成功信息
    printf("挂号成功！挂号编号：%s，患者：%s，医生：%s，科室：%s，日期：%s\n",
        newNode->data.regNo, newNode->data.patientName,
        newNode->data.doctorName, newNode->data.dept, newNode->data.date);
    
    return 1;
}
```

### 3.3 取消挂号
```c
int cancelRegistration(Registration** head, Registration** tail,
    Registration* r) {
    if (r == NULL) {
        printf("未找到对应记录\n");
        return 0;
    }
    
    // 检查挂号状态，只有"待就诊"状态的挂号才能取消
    if (strcmp(r->data.status, "待就诊") != 0) {
        printf("只有待就诊状态的挂号才能取消\n");
        return 0;
    }
    
    // 查找对应医生并减少其当前患者数量
    Doctor* d = findDoctorByEmpNo(getDoctorHead(), r->data.doctorEmpNo);
    if (d != NULL && d->data.currentPatients > 0) {
        d->data.currentPatients--;
    }
    
    // 修改挂号记录状态为"已取消"
    strcpy(r->data.status, "已取消");
    
    // 更新挂号记录文件
    rebuildRegistrationFile(*head);
    printf("取消成功\n");
    return 1;
}
```

### 3.4 完成就诊
```c
int completeRegistration(Registration* r) {
    if (r == NULL) {
        printf("挂号信息不能为空\n");
        return 0;
    }
    
    // 检查挂号状态是否为"待就诊"或"就诊中"
    if (strcmp(r->data.status, "待就诊") != 0 && 
        strcmp(r->data.status, "就诊中") != 0) {
        printf("挂号信息状态异常，无法完成就诊\n");
        return 0;
    }
    
    // 修改状态为"已完成"
    strcpy(r->data.status, "已完成");
    printf("患者%s就诊完成\n", r->data.patientName);
    return 1;
}
```

### 3.5 叫号功能（医生端）
```c
int callNextPatient(Registration** head, Registration** tail, char* doctorEmpNo) {
    if (head == NULL || *head == NULL) {
        printf("暂无挂号记录\n");
        return 0;
    }
    
    Registration* current = *head;
    while (current != NULL) {
        // 查找指定医生的待就诊患者
        if (strcmp(current->data.doctorEmpNo, doctorEmpNo) == 0 &&
            strcmp(current->data.status, "待就诊") == 0) {
            
            // 修改状态为"就诊中"
            strcpy(current->data.status, "就诊中");
            printf("请患者 %s 到 %s 就诊\n", current->data.patientName, 
                   current->data.doctorName);
            
            // 更新文件
            rebuildRegistrationFile(*head);
            return 1;
        }
        current = current->next;
    }
    
    printf("该医生暂无待就诊患者\n");
    return 0;
}
```

#### 3.6.4 按日期范围查询
```c
Registration* findRegistrationsByDateRange(Registration* head, int y1,
    int m1, int d1, int y2, int m2, int d2) {
    Registration* r = head;
    Registration* resultHead = NULL;
    Registration* resultTail = NULL;
    
    while (r != NULL) {
        int regYear, regMonth, regDay;
        sscanf(r->data.date, "%d-%d-%d", &regYear, &regMonth, &regDay);
        
        // 比较日期范围
        if (compareDate(regYear, regMonth, regDay, y1, m1, d1) >= 0 &&
            compareDate(regYear, regMonth, regDay, y2, m2, d2) <= 0) {
            
            // 创建新节点并复制数据
            Registration* newNode = (Registration*)malloc(sizeof(Registration));
            if (newNode != NULL) {
                newNode->data = r->data;
                newNode->next = NULL;
                newNode->pre = NULL;
                
                if (resultHead == NULL) {
                    resultHead = newNode;
                    resultTail = newNode;
                } else {
                    resultTail->next = newNode;
                    newNode->pre = resultTail;
                    resultTail = newNode;
                }
            }
        }
        r = r->next;
    }
    return resultHead;
}
```

```c
void rebuildRegistrationFile(Registration* head) {
    FILE* fp = fopen(REGISTRATION_FILE, "w");
    if (fp == NULL) {
        printf("无法打开挂号记录文件进行写入\n");
        return;
    }
    
    Registration* p = head;
    while (p != NULL) {
        fprintf(fp, "1\n");
        fprintf(fp, "%s\n", p->data.regNo);
        fprintf(fp, "%s\n", p->data.patientCardNo);
        fprintf(fp, "%s\n", p->data.patientName);
        fprintf(fp, "%s\n", p->data.doctorEmpNo);
        fprintf(fp, "%s\n", p->data.doctorName);
        fprintf(fp, "%s\n", p->data.dept);
        fprintf(fp, "%s\n", p->data.date);
        fprintf(fp, "%s\n", p->data.time);
        fprintf(fp, "%s\n", p->data.status);
        fprintf(fp, "\n");  // 空行分隔
        p = p->next;
    }
    fprintf(fp, "-1\n");
    fclose(fp);
}
```
